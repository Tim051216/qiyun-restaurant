## 生成测试数据

### 方法一：使用云对象测试工具

#### 1. 单个生成

1. 在 HBuilderX 中打开 `table-qrcode/index.obj.js`
2. 右键选择"运行云函数"
3. 输入测试参数：

```json
{
  "method": "generateQRCode",
  "params": ["A01"]
}
```

4. 查看返回结果：

```json
{
  "success": true,
  "data": {
    "tableNumber": "A01",
    "qrcodeUrl": "cloud://xxx.png",
    "createTime": 1234567890
  }
}
```

#### 2. 批量生成脚本

创建测试脚本 `generate-test-tables.js`：

```javascript
// 在云函数中运行此脚本
const tableNumbers = [
  'A01', 'A02', 'A03', 'A04', 'A05',
  'B01', 'B02', 'B03', 'B04', 'B05',
  'V01', 'V02', 'P01', 'P02'
]

async function generateTestTables() {
  const results = []
  
  for (const tableNumber of tableNumbers) {
    try {
      console.log(`生成桌号: ${tableNumber}`)
      
      const result = await this.generateQRCode(tableNumber)
      
      if (result.success) {
        console.log(`✓ ${tableNumber} 生成成功`)
        results.push({
          tableNumber,
          success: true,
          qrcodeUrl: result.data.qrcodeUrl
        })
      } else {
        console.error(`✗ ${tableNumber} 生成失败: ${result.errMsg}`)
        results.push({
          tableNumber,
          success: false,
          error: result.errMsg
        })
      }
      
      // 延迟避免频率限制
      await new Promise(resolve => setTimeout(resolve, 1000))
    } catch (error) {
      console.error(`✗ ${tableNumber} 异常: ${error.message}`)
      results.push({
        tableNumber,
        success: false,
        error: error.message
      })
    }
  }
  
  return results
}

// 执行生成
generateTestTables().then(results => {
  console.log('生成完成，结果：')
  console.log(JSON.stringify(results, null, 2))
})
```
